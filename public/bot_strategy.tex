\begin{abstract}
We document the exact mathematical invariants and search algorithms used by the artificial players (\emph{Bots}) and selected-coin \emph{Best-Move Hint} engine in \textbf{LineLimit 45} (v2.2.1) across both \textbf{Adversarial} (\texttt{ADV}) and \textbf{Cooperative} (\texttt{CO-OP}) modes. Both modes share a fast $O(1)/O(n^2)$ combinatorial core built on: (1)~the \emph{Zero-Slack Lock-Floor Theorem}, (2)~the \emph{Draw $\Longleftrightarrow$ Incremental Latin Square Theorem}, (3)~the \emph{$90^\circ$ Non-Attacking Quartet Theorem} and $\gcd(n,6)=1$ \emph{toroidal knight transversals}, (4)~global and per-row/per-column \emph{Latin sublevel deficit} bounds, and (5)~an \emph{Exact Ceiling-Frozen Endgame Shortcut}. In \texttt{ADV}, the Bot combines iterative-deepening alpha-beta Negamax with $[v_{\max}]\,\texttt{x1}$ trapdoor-cliff parity control, dynamic $90^\circ$/$180^\circ$ orbit completion, odd-$n$ center seizure, escalating Main-X mirrors ($w \to w+1$), and Boltzmann band sampling. In \texttt{CO-OP}, an expanded verified blueprint catalog, per-row/per-column Hall feasibility checks, and exact Minimum-Remaining-Values (MRV) backtracking guarantee reliable full-board completion all the way up to $12 \times 12$. Cooperative Bots \emph{never communicate}: coordination emerges purely from the visible board.
\end{abstract}

\section{Game State, Legality, and Scoring}

Let $n \in \{4, \dots, 12\}$ be the board dimension. The board has $n^2$ cells $(r,c) \in \{0,\dots,n-1\}^2$ and a shared multiset of $n^2$ coins containing $n$ copies of each value $v \in \{1,\dots,n\}$. Let $\mathrm{rem}[v] \in \{0,\dots,n\}$ denote the remaining supply of value $v$, and let $\mathcal{L}(r,c)$ be the set of $2$ to $4$ lines (row $r$, column $c$, and any $45^\circ$ diagonal of length $\ge 2$) passing through $(r,c)$.

\begin{definition}[Line Safety and Threshold-Sum Scoring]
A board is \emph{safe} iff for every line $L$ and every coin value $v$ appearing on $L$,
\[
  \mathrm{count}_L(\le v) := |\{x \in L : \mathrm{val}(x) \le v\}| \le v.
\]
When a move causes at least one line violation ($\mathrm{count}_L(\le v) > v$), the player who placed the coin loses immediately. On each violated line $L$, let
\[
  v_{\max}(L) := \max\{v : \mathrm{count}_L(\le v) > v\}.
\]
The winner's score is the sum of all distinct board coins in $\bigcup_{L \text{ violated}} \{x \in L : \mathrm{val}(x) \le v_{\max}(L)\}$ (each cell counted once). In \texttt{CO-OP}, both players form a team that wins iff all $n^2$ cells are filled with zero violations.
\end{definition}

\section{Exact Combinatorial Theorems in the Engine}

\subsection{\texorpdfstring{Zero-Slack Lock-Floor Theorem ($O(1)$ Legality)}{Zero-Slack Lock-Floor Theorem (O(1) Legality)}}

For each line $L$ of a currently safe board, let $f_L(v)$ be the multiplicity of coin $v$ on $L$ and $\mathrm{cum}_L(v) = \sum_{w=1}^{v} f_L(w)$.

\begin{theorem}[Lock Floor]
Define the zero-slack lock threshold of line $L$ by
\[
  z(L) := \max\bigl(\{0\} \cup \{v \in \{1,\dots,n\} : f_L(v) > 0 \text{ and } \mathrm{cum}_L(v) = v\}\bigr).
\]
Then placing coin $v$ on an empty cell $(r,c)$ keeps the board safe iff
\[
  v \ge F(r,c) := 1 + \max_{L \in \mathcal{L}(r,c)} z(L).
\]
\end{theorem}

By maintaining $f_L(v)$ and $z(L)$ incrementally, legality of $(r,c,v)$ is tested in $O(1)$ time, and the exact total count of safe moves across the board, $\mathrm{totalSafe}$, is computed in $O(n^2)$ time via $\sum_{(r,c)\text{ empty}} \mathrm{availGeq}[F(r,c)]$.

\subsection{\texorpdfstring{Draw $\Longleftrightarrow$ Incremental Latin Square}{Draw <=> Incremental Latin Square}}

\begin{theorem}[Latin Necessity of Full Boards]\label{thm:draw-latin}
For every $n \ge 4$, a completely filled $n \times n$ board is free of violations if and only if it is an incremental $n \times n$ Latin square (every row and column is a permutation of $\{1,\dots,n\}$).
\end{theorem}

\begin{proof}
By induction on $v = 1,\dots,n$. No row or column can hold two $1$s ($\mathrm{count}(\le 1) \ge 2 > 1$); since $n$ copies of $1$ are placed across $n$ rows and $n$ columns, each row and column holds exactly one $1$. Inductively, each row and column already contains $v-1$ coins strictly smaller than $v$; placing two copies of $v$ in the same row or column yields $\mathrm{count}(\le v) \ge (v-1)+2 = v+1 > v$, a violation. Hence all $n$ copies of $v$ distribute one per row and one per column.
\end{proof}

\begin{corollary}[Instant Draw-Destruction]\label{cor:draw-death}
If any row or column contains two equal coins (even two copies of $n$), or on $n=4$ if coin $1$ occupies any corner or center cell (absent in all $16$ incremental $4\times 4$ Latin squares), zero full-board Draws remain reachable: at least one empty cell must become dead (\texttt{KO}) before ply $n^2$.
\end{corollary}

\subsection{\texorpdfstring{The $90^\circ$ Non-Attacking Quartet Theorem and Toroidal Transversals}{The 90-Degree Non-Attacking Quartet Theorem and Toroidal Transversals}}

\begin{theorem}[$90^\circ$ Quartet Independence]\label{thm:quartet}
Let $n$ be even and let $(r,c)$ lie strictly off the Main X ($r \ne c$ and $r+c \ne n-1$). Then its $90^\circ$ rotational orbit
\[
  O_{90}(r,c) = \bigl\{(r,c),\; (c,n-1-r),\; (n-1-r,n-1-c),\; (n-1-c,r)\bigr\}
\]
consists of $4$ pairwise non-attacking queens (four distinct rows, four distinct columns, four distinct $\setminus$ diagonals, and four distinct $/$ diagonals). Conversely, on $n=4$, the center orbit and corner orbit on the Main X each form complete conflict cliques $K_4$.
\end{theorem}

For odd $n$ coprime to $6$ ($n \in \{5,7,11\}$), toroidal knight-steps $((r \pm 1)\bmod n, (c \pm 2)\bmod n)$ and $((r \pm 2)\bmod n, (c \pm 1)\bmod n)$ similarly generate $n$-cell non-attacking Latin transversals.

\subsection{Dead Squares, Sublevel Deficit, and Trapdoor Cliffs}

Let $v_{\mathrm{top}} = \max(\{0\} \cup \{v : \mathrm{rem}[v] > 0\})$ and $v_{\mathrm{sec}} = \max(\{0\} \cup \{v < v_{\mathrm{top}} : \mathrm{rem}[v] > 0\})$.
\begin{itemize}[nosep]
  \item A cell $(r,c)$ is \textbf{dead} (\texttt{KO}) iff $F(r,c) > v_{\mathrm{top}}$. While $\mathrm{rem}[n] > 0$, $|L| \le n \implies z(L) \le n-1 \implies F(r,c) \le n$, so zero \texttt{KO} cells can exist and placing coin $n$ \emph{never} raises any other empty cell's floor (\textbf{Zero-Splash Property}).
  \item Let $R_{\le k} = \sum_{v=1}^{k} \mathrm{rem}[v]$ and $S_{\le k} = |\{(r,c)\text{ empty} : F(r,c) \le k\}|$. If $\max_{1 \le k \le n}(R_{\le k} - S_{\le k}) > 0$, no full-board completion exists (\textbf{Pigeonhole Sublevel Deficit}).
  \item The \textbf{Trapdoor Cliff} $K_{\mathrm{cliff}} = |\{(r,c)\text{ empty} : v_{\mathrm{sec}} < F(r,c) \le v_{\mathrm{top}}\}|$ counts squares kept alive solely by $v_{\mathrm{top}}$. Spending the final copy ($\mathrm{rem}[v_{\mathrm{top}}] = 1 \to 0$) reduces playable squares by $1 + K_{\mathrm{cliff}}$ on a single turn.
\end{itemize}

\subsection{Exact Ceiling-Frozen Endgame Shortcut}

\begin{proposition}[Frozen-Ceiling Tablebase]
Let $P$ be the count of playable cells ($F(r,c) \le v_{\mathrm{top}}$) and $F_{\min} = \min_{(r,c)\text{ empty}} F(r,c)$. Suppose $v_{\mathrm{sec}} < F_{\min}$ (so only $v_{\mathrm{top}}$ is ever playable), and either $v_{\mathrm{top}} = n$ or for every line $L$ with $p_L \ge 2$ playable cells and every $w \in \{v_{\mathrm{top}},\dots,n-1\}$ ($w = v_{\mathrm{top}}$ or $f_L(w) > 0$),
\[
  \mathrm{cum}_L(w) + \min(p_L - 1,\; \mathrm{rem}[v_{\mathrm{top}}]) < w.
\]
Then no move can ever lock another playable square, and every play sequence lasts \emph{exactly} $M = \min(P,\; \mathrm{rem}[v_{\mathrm{top}}])$ plies: if $M$ equals the total empty cells, the position is an exact Draw ($0$); otherwise the side to move wins in $M$ plies iff $M$ is odd.
\end{proposition}

Evaluating this condition in $O(n^2)$ inside \texttt{negamax} prunes up to $24$ plies of endgame permutations instantaneously, even on $12 \times 12$.

\section{\texorpdfstring{Adversarial Mode (\texttt{ADV}) Strategy}{Adversarial Mode (ADV) Strategy}}

If $\mathrm{safeMoves} = \emptyset$, the Bot deterministically picks the legal move minimizing $\mathrm{totalWinningsIfLost}$. Otherwise, \texttt{ADV} runs iterative-deepening Alpha-Beta Negamax ($T = 95\,\mathrm{ms}$, $60{,}000$ nodes, beam width $36$, root beam $72$, TT cap $120{,}000$):

\begin{enumerate}[nosep]
  \item \textbf{1-Ply Win Scan}: Any safe move leaving $\mathrm{oppSafe} = 0$ is immediately flagged as a forced win ($+100{,}000$).
  \item \textbf{Symmetry, Orbit, Cliff, and Squeeze Ordering}: Root candidates $(r,c,v)$ combine opponent squeeze ($-4\,\mathrm{oppSafe}$), line multiplicity ($2.5|\mathcal{L}(r,c)| - 0.4\,d_{\mathrm{center}}$), tight floor locks ($v = F(r,c)$ with $v \le 2$), zero-splash cleanup ($v = n$ on $F = n$, $+5$), $[v_{\mathrm{top}}]\,\texttt{x1}$ cliff parity ($+10$ if leaving an even playable count, $-8$ if odd), $\gcd(n,6)=1$ knight-transversal continuity ($+3.5$), and $D_4(L^n)$ compatibility ($+4$). In addition:
  \begin{itemize}[nosep]
    \item \emph{Opening \& Odd-$n$ ``Steal the Center''}: On odd $n$, P1 receives $+5$ for opening at $(\frac{n-1}{2},\frac{n-1}{2})$; if P1 leaves the center open on ply $\ge 1$, P2 receives $+7.5$ to immediately seize $(\frac{n-1}{2},\frac{n-1}{2})$ with $v = F \le 3$ and flip symmetric parity. On even $n$, P1 receives $+4.5$ for small Main-X openings that preempt rotational mirroring.
    \item \emph{Dynamic $90^\circ$ Quartet Completion}: For P2 on even $n$ off the Main X (Theorem~\ref{thm:quartet}), whenever $O_{90}(r,c)$ currently holds $1$ or $3$ copies of $v$ while $\mathrm{rem}[v]$ is odd, playing $v$ in $(r,c)$ receives $+7$ to $+8$, restoring both orbit and stack parity across up to $3$ varied non-attacking cells (with $+5.5$ off the cross/Main-X on odd $n$).
    \item \emph{Escalating Mirror ($w \to w+1$)}: When an opponent places coin $w$ on the Main X or odd cross so a rotational partner $(r,c)$ has floor raised to $F(r,c) = w+1$, answering on $(r,c)$ with $v = w+1$ receives $+6$, locking the shared line at $z(L) = w+1$.
  \end{itemize}
  \item \textbf{Phase-Scaled Leaf Evaluation}: At depth $0$ (after checking the exact Frozen-Ceiling Tablebase), let $\mathrm{drawsDead}$ combine dead cells, sublevel deficit, and Corollary~\ref{cor:draw-death}, and let
  \[
    \mathrm{effPlayable} = \min(P,\; R_{\mathrm{total}} - \max_k(R_{\le k} - S_{\le k})) - \mathbf{1}_{\mathrm{drawsDead} \land \text{full}}.
  \]
  As the fraction $\rho \in [0,1]$ of playable cells at floors $\ge v_{\mathrm{top}}-1$ approaches $1$, the odd/even parity weight on $\mathrm{effPlayable}$ scales from $\pm 20$ up to $\pm 65$ while mobility weight tapers by $1 - 0.55\rho$, supplemented by a $+24$ bonus when holding $\mathrm{rem}[v_{\mathrm{top}}]=1$ with odd $K_{\mathrm{cliff}}$ (two moves of opposite $\Delta P$ parities).
  \item \textbf{Maximal Non-Transparent Play Diversity}: Candidate arrays are Fisher--Yates shuffled before sorting to eliminate row-major top-left bias, a single compatible $D_4(L^n)$ template is chosen at random per move, immediate/forced wins within $4$ plies are sampled uniformly, and non-losing candidates (spread $14\text{--}22$ points, up to $28$ candidates) are sampled via phase-annealed Boltzmann weights ($\tau \in [5.0, 8.5]$).
\end{enumerate}

\section{\texorpdfstring{Cooperative Mode (\texttt{CO-OP}) Strategy \& Implicit Coordination}{Cooperative Mode (CO-OP) Strategy and Implicit Coordination}}

In \texttt{CO-OP}, the Bot evaluates every safe move using three exact feasibility layers:
\begin{enumerate}[nosep]
  \item \textbf{Global \& Per-Line Latin Feasibility ($O(n^2)$)}: Penalizes \texttt{KO} cells ($-50{,}000+$), global sublevel deficits ($-40{,}000+$), row/column duplicates or $4\times 4$ forbidden-$1$s ($-45{,}000$), and \emph{per-row / per-column Latin deficits} ($-38{,}000+$ whenever any missing value $v$ in row $r$ or column $c$ lacks a crossing-compatible empty cell with $F \le v$, or missing values $\le k$ in that row/column exceed empty cells with $F \le k$).
  \item \textbf{Implicit Blueprint Tier Beyond the MRV Horizon ($> 30$ Empty Cells)}: An expanded cached pool of verified zero-violation full boards (all $16$ solutions on $4\times 4$, and toroidal shifts $\times$ $D_4$ symmetries up to $88$ boards on larger $n$) is filtered against the current visible board. When $\mathrm{emptyCount} > 30$ and consistent blueprints exist, one compatible blueprint is sampled per turn and granted a $+2{,}500$ tier bonus while exploration jitter ($\pm 13$) and headroom scoring randomize the move sequence across up to $(n^2)!$ orderings---preventing midgame $n$-queens deadlocks even on $12 \times 12$.
  \item \textbf{Exact MRV Completion ($\le 30$ Empty Cells)}: Shuffled healthy candidates (up to $28$) are verified via an exact \textbf{Minimum-Remaining-Values} solver (\texttt{canComplete\allowbreak CoopBoard}, $75\,\mathrm{ms}$ shared budget, $5{,}000$ nodes per candidate) with row/column Hall deficit pruning at every node, sampling \emph{uniformly across all MRV-completable moves}.
\end{enumerate}

\textbf{Do Co-op Bots Communicate?} \textbf{No.} Both \texttt{ADV} and \texttt{CO-OP} Bots are completely stateless functions of the visible board alone; no target Latin square, history, or hidden message is ever stored or passed between turns.

\section{\texorpdfstring{Varied Best-Move Hint (\texttt{HINTS} $\to$ \texttt{DOT})}{Varied Best-Move Hint (HINTS -> DOT)}}

Clicking the \texttt{HINTS} gadget cycles through three display states: \emph{off} $\to$ \emph{cell safety hints} $\to$ \emph{best-move dot} (\texttt{DOT~$\bullet$}) $\to$ \emph{off}. In the \texttt{DOT~$\bullet$} state, \texttt{getBestHintMove(\allowbreak board, \allowbreak rule, \allowbreak selectedCoin)} runs the varied \texttt{ADV} or \texttt{CO-OP} pipeline ($65\,\mathrm{ms}$ budget) restricted to root placements of the player's currently selected coin $v_{\mathrm{sel}}$, strictly rejecting both immediate violations ($v_{\mathrm{sel}} < F(r,c)$) and lookahead-proven losing moves ($\mathrm{score} \le -\mathrm{WIN\_THRESHOLD}$ in \texttt{ADV}, or uncompletable/dead states in \texttt{CO-OP}). If a strong non-losing square $(r^\star, c^\star)$ exists for $v_{\mathrm{sel}}$, it is marked with a centered monochrome dot; if every placement of $v_{\mathrm{sel}}$ leads to a loss, no board dot is shown (and if another coin $v'$ admits a non-losing move, a subtle dot marks $[v']$ in the tray).
